# command guard — the denylist patterns that read a secret.
# Each line is a verbatim copy of a line in dangerous-patterns.txt; test-guard.sh
# fails when one is not. A command matching one is explained to the user, never
# handed over: running it would put the secret into the conversation.
# A separate list, so upstream's groups in dangerous-patterns.txt stay unedited.

# 9. gh token exfil
(^|[;&|[:space:]])gh[[:space:]]+auth[[:space:]]+token([[:space:]]|$)

# 10. password managers & OS keychain, minus opening or uninstalling the app
(^|[;&|[:space:]])(bw|bws|lpass|keepassxc-cli|rbw|nordpass)([[:space:]]|$)
(^|[;&|])[[:space:]]*pass[[:space:]]+[^[:space:]]
(^|[;&|[:space:]])op[[:space:]]+(read|run|inject|item|document|vault|connect|service-account|events-api|signin)([[:space:]]|$)
(^|[;&|[:space:]])security[[:space:]]+((-[a-zA-Z]+|--[a-z-]+)[[:space:]]+)*(find-generic-password|find-internet-password|dump-keychain)([[:space:]]|$)
(1Password|Bitwarden|NordPass|KeePassXC)\.app
(~|\$HOME|\$\{HOME\}|/Users/[^/[:space:]"']+)/\.password-store
--export-secret-(keys?|subkeys)([[:space:]=]|$)

# 11. glab token exfil
(^|[;&|[:space:]])glab[[:space:]]+auth[[:space:]]+token([[:space:]]|$)

# 12. cloud reads that hand over a credential, all of them
(^|[;&|[:space:]])aws[[:space:]][^;&|]*[[:space:]]([a-z0-9-]*-)?get-[a-z0-9-]*(token|password|credential|secret|login)
(^|[;&|[:space:]])aws[[:space:]]([^;&|]*[[:space:]])?ssm[[:space:]][^;&|]*--with-decryption([[:space:]]|$)
(^|[;&|[:space:]])aws[[:space:]]([^;&|]*[[:space:]])?lightsail[[:space:]]+get-instance-access-details([[:space:]]|$)
(^|[;&|[:space:]])aws[[:space:]]([^;&|]*[[:space:]])?--debug([[:space:]]|$)
(^|[;&|[:space:]])gcloud[[:space:]][^;&|]*[[:space:]]get-credentials([[:space:]]|$)
(^|[;&|[:space:]])gcloud[[:space:]][^;&|]*[[:space:]]print-(access|identity)-token([[:space:]]|$)
(^|[;&|[:space:]])gcloud[[:space:]]([^;&|]*[[:space:]])?secrets[[:space:]]+versions[[:space:]]+access([[:space:]]|$)
(^|[;&|[:space:]])aws[[:space:]]([^;&|]*[[:space:]])?configure[[:space:]]+(get[[:space:]][^;&|]*(secret|token)|export-credentials)
(^|[;&|[:space:]])aws[[:space:]]([^;&|]*[[:space:]])?sts[[:space:]]+assume-role[a-z-]*([[:space:]]|$)
